    '/content/drive/MyDrive/data'
]
if not os.path.exists('data/clean') or len(os.listdir('data/clean')) == 0:
    for dpath in drive_data_candidates:
        if os.path.exists(os.path.join(dpath, 'clean')):
            print(f"⚡ Copying dataset from Google Drive ({dpath}) to fast NVMe SSD (~2-3 mins once)...")
            os.makedirs('data', exist_ok=True)
            !cp -r "{dpath}/clean" data/
            !cp -r "{dpath}/stego" data/
